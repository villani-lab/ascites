DC_palette <- c("cDC1" = '#B8ABE0', "cDC2" = '#7F3F98', "DC: PIGR, RORC" = '#FFC857', "CD14" = "#36565C")

CFSE_data <- read.csv("/projects/home/tlchan/projects/ascites/figure_panels/data/fig_5_data/combined_CFSE_stats.csv") %>%
    mutate(cell_category = factor(cell_category, levels = c("CD14", "DC: PIGR, RORC", "cDC1", "cDC2")))

stats <- CFSE_data %>%
    filter(cell_category %in% c('DC: PIGR, RORC', 'CD14')) %>%
    t_test(log_CFSE_neg_perc ~ cell_category, paired = TRUE) %>%
    add_significance() %>%
    add_xy_position(x = "cell_category") %>%
    mutate(p_label = paste0("p = ", round(p, 3)))

stats$clean_pvals <- sapply(stats$p, function(x) {
    if (x > 0.01) {
        return(as.character(round(x, 2)))
    } else if (x < 0.01 & x > 0.001) {
        return(as.character(round(x, 3)))
    } else {
        formatC(x, format = "e", digits = 0)
    }
})

ggplot(CFSE_data, aes(x = factor(cell_category), y = log_CFSE_neg_perc)) +
    geom_jitter(pch = 21, size = 5, width = 0.1, height = 0, aes(fill = cell_category)) +
    xlab("Cell category") +
    ylab("log(CFSE- percentage)") +
    scale_fill_manual(values = DC_palette) +
    stat_pvalue_manual(data = stats, label = "p = {clean_pvals}", size = 7) +
    theme_classic(base_size = 22) +
    theme(axis.text.x = element_text(angle = 90, hjust = 1), legend.position = "none")

cd4_prolif_data <- read.csv('/projects/home/tlchan/projects/ascites/figure_panels/data/violinplot_data/spatial_cd4_prolif_gene_exp.csv') %>%
    mutate(DC_RORC_niche = ifelse(DC_RORC_niche == "CD4_prolif_non_niche", "Outside DC_RORC niche", "Inside DC_RORC niche")) %>%
    mutate(DC_RORC_niche = factor(DC_RORC_niche, levels = c("Outside DC_RORC niche", "Inside DC_RORC niche"))) %>%
    mutate(Gene = factor(Gene, levels = c("CD3E", "CD4", "IFNG", "IL5", "IL13")))

stats <- cd4_prolif_data %>%
    group_by(Gene) %>%
    t_test(Count ~ DC_RORC_niche) %>%
    adjust_pvalue(method = "fdr") %>%
    add_significance() %>%
    add_xy_position(x = "DC_RORC_niche") %>%
    mutate(text_color = ifelse(p.adj < 0.05, "black", "gray"))

stats$clean_pvals <- sapply(stats$p, function(x) {
    if (x > 0.01) {
        return(as.character(round(x, 2)))
    } else if (x < 0.01 & x > 0.001) {
        return(as.character(round(x, 3)))
    } else {
        formatC(x, format = "e", digits = 0)
    }
})

ggplot(cd4_prolif_data, aes(x = DC_RORC_niche, y = Count)) +
    geom_violin(aes(fill = DC_RORC_niche), alpha = 0.75) +
    geom_jitter(width = 0.1, size = .5) +
    facet_wrap("Gene", scales = "free_y") +
    theme_classic(base_size = 20) +
    scale_fill_manual(values = niche_palette, guide = "none") +
    scale_y_continuous(expand = expansion(mult = c(0.05, 0.15))) +
    stat_pvalue_manual(data = stats, label = "p = {clean_pvals}", size = 6, color = "text_color") +
    scale_color_manual(values = c("black" = "#000000", "gray" = "#808080"), guide = "none") +
    theme(strip.text = element_text(face = "italic"), axis.text.x = element_blank())
